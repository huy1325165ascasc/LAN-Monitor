    @GetMapping
    public ResponseEntity<List<ClientSession>> getClients() {
        return ResponseEntity.ok(clientManager.getAllClients());
    }

    @PostMapping("/{clientId}/capture")
    public ResponseEntity<?> requestScreenshot(@PathVariable String clientId) {
        ClientConnectionHandler handler = clientManager.getHandler(clientId);
        if (handler != null) {
            handler.sendRawCommand("CAPTURE_SCREEN");
            return ResponseEntity.ok(Map.of("message", "Đã gửi lệnh chụp màn hình"));
        }
        return ResponseEntity.badRequest().body("Client không tồn tại hoặc offline");
    }

    @PostMapping("/{clientId}/alert")
    public ResponseEntity<?> sendAlert(
            @PathVariable String clientId,
            @RequestBody Map<String, String> body) {
        ClientConnectionHandler handler = clientManager.getHandler(clientId);
        if (handler != null) {
            String message = body.getOrDefault("message", "");
            handler.sendRawCommand("ALERT:" + message);
            return ResponseEntity.ok(Map.of("message", "Đã gửi cảnh báo thành công"));
        }
        return ResponseEntity.badRequest().body("Client không tồn tại hoặc offline");
    }

    @PostMapping("/{clientId}/save-frame")
    public ResponseEntity<?> saveLivestreamFrame(@PathVariable String clientId) {
        ClientConnectionHandler handler = clientManager.getHandler(clientId);
        if (handler != null) {
            handler.setCaptureNextFrame(true); // Bật cờ chụp ở Handler
            return ResponseEntity.ok(Map.of("message", "Đã yêu cầu lưu ảnh Livestream"));
        }
        return ResponseEntity.badRequest().body("Client không tồn tại hoặc offline");
    }

}

package com.vku.lanmonitor.server.controller;

import com.fasterxml.jackson.databind.ObjectMapper;
import com.vku.lanmonitor.server.socket.ClientManager;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;

import java.io.File;
import java.nio.file.Files;
import java.nio.file.Paths;
import java.util.Arrays;
import java.util.List;
import java.util.Map;

@Slf4j
@RestController
@RequestMapping("/api/settings")
@RequiredArgsConstructor
public class SettingsController {

    private final ClientManager clientManager;
    private final ObjectMapper objectMapper = new ObjectMapper();
    private final String STORAGE_DIR = "storage";
    private final String RULES_FILE = STORAGE_DIR + "/exam_rules.json";

    // 1. Lấy danh sách Whitelist hiện tại đang lưu
    @GetMapping("/whitelist")
    public ResponseEntity<List<String>> getWhitelist() {
        try {
            File file = new File(RULES_FILE);
            if (file.exists()) {
                List<String> whitelist = Arrays.asList(objectMapper.readValue(file, String[].class));
                return ResponseEntity.ok(whitelist);
            }
        } catch (Exception e) {
            log.error("Lỗi đọc file cấu hình: {}", e.getMessage());
        }
        // Trả về mặc định nếu chưa có file
        return ResponseEntity.ok(Arrays.asList("Bài Thi Trắc Nghiệm", "Word", "Calculator", "Excel"));
    }

    // 2. Cập nhật Whitelist mới từ Web Dashboard
    @PostMapping("/whitelist")
    public ResponseEntity<?> updateWhitelist(@RequestBody Map<String, List<String>> body) {
        List<String> newWhitelist = body.get("whitelist");
        if (newWhitelist == null || newWhitelist.isEmpty()) {
            return ResponseEntity.badRequest().body("Danh sách trắng không được để trống");
        }

        try {
            // Đảm bảo thư mục storage tồn tại
            Files.createDirectories(Paths.get(STORAGE_DIR));

            // Lưu danh sách mới vào file JSON
            objectMapper.writeValue(new File(RULES_FILE), newWhitelist);
            log.info("Đã lưu cấu hình Whitelist mới vào: {}", RULES_FILE);

            // Chuyển mảng thành chuỗi cách nhau bằng dấu phẩy (VD: "Word,Excel,Calculator")
            String joinedKeywords = String.join(",", newWhitelist);

            // Phát sóng lệnh xuống toàn bộ máy Client đang bật
            clientManager.broadcastCommand("UPDATE_WHITELIST:" + joinedKeywords);

            return ResponseEntity.ok(Map.of("message", "Đã cập nhật danh sách và gửi lệnh đến các máy trạm!"));
        } catch (Exception e) {
            log.error("Lỗi khi ghi file cấu hình: {}", e.getMessage());
            return ResponseEntity.internalServerError().body("Lỗi Server: " + e.getMessage());
        }
    }
}

package com.vku.lanmonitor.server.socket;

import com.vku.lanmonitor.server.model.ClientSession;
import com.vku.lanmonitor.server.websocket.RealtimeNotificationService;
import lombok.extern.slf4j.Slf4j;

import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.net.Socket;
import java.nio.charset.StandardCharsets;

import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.util.Base64;

@Slf4j
public class ClientConnectionHandler implements Runnable {

    private final ClientSession session;
    private final ClientManager clientManager;
    private final RealtimeNotificationService notificationService;
    private PrintWriter out;
    private BufferedReader in;

    private volatile boolean captureNextFrame = false;

    public void setCaptureNextFrame(boolean capture) {
        this.captureNextFrame = capture;
    }

    public ClientConnectionHandler(ClientSession session, ClientManager clientManager,
            RealtimeNotificationService notificationService) {
        this.session = session;
        this.clientManager = clientManager;
        this.notificationService = notificationService;
    }

    @Override
    public void run() {
        Socket socket = session.getSocket();
        try {
            out = new PrintWriter(socket.getOutputStream(), true, StandardCharsets.UTF_8);
            in = new BufferedReader(new InputStreamReader(socket.getInputStream(), StandardCharsets.UTF_8));
            notificationService.notifyClientListUpdated(clientManager.getAllClients());

            // Tự động tạo 2 thư mục nếu chưa tồn tại
            Files.createDirectories(Paths.get("images"));
            Files.createDirectories(Paths.get("temp_images"));

            // Nạp luật thi ngay khi máy sinh viên vừa kết nối
            try {
                java.io.File ruleFile = new java.io.File("storage/exam_rules.json");
                if (ruleFile.exists()) {
                    com.fasterxml.jackson.databind.ObjectMapper mapper = new com.fasterxml.jackson.databind.ObjectMapper();
                    String[] savedRules = mapper.readValue(ruleFile, String[].class);
                    String joinedRules = String.join(",", savedRules);
                    out.println("UPDATE_WHITELIST:" + joinedRules);
                    log.info("Đã đồng bộ luật thi (Whitelist) cho Client: {}", session.getId());
                }
            } catch (Exception e) {
                log.warn("Chưa có luật thi, Client {} sẽ chờ cấu hình từ Web.", session.getId());
            }

            String line;
            while (session.isOnline() && (line = in.readLine()) != null) {

                // 1. LUỒNG LIVESTREAM
                if (line.startsWith("SCREEN:")) {
                    String base64Image = line.substring(7);
                    notificationService.sendScreenshotToWeb(session.getId(), base64Image);

                    // NẾU GIÁM THỊ VỪA BẤM NÚT CHỤP ẢNH -> LƯU LẠI KHUNG HÌNH NÀY
                    if (captureNextFrame) {
                        captureNextFrame = false; // Tắt cờ
                        String fileName = "manual_" + session.getId().replace(":", "_") + "_"
                                + System.currentTimeMillis() + ".jpg";
                        Path imagePath = Paths.get("images", fileName);
                        byte[] imageBytes = Base64.getDecoder().decode(base64Image);
                        Files.write(imagePath, imageBytes);
                        log.info("Đã lưu khung hình Livestream thành công: {}", fileName);
                    }
                }

                // 2. LUỒNG BẰNG CHỨNG (Giờ chỉ nhận Text để hiển thị, không lưu file)
                else if (line.startsWith("SUSPICIOUS:")) {
                    String[] parts = line.substring(11).split("\\|", 2); // Chỉ cắt làm 2 phần
                    if (parts.length == 2) {
                        String severity = parts[0];
                        String tabName = parts[1];

                        // imageUrl truyền vào là null vì không có ảnh
                        notificationService.sendSuspiciousAlertToWeb(
                                session.getId(), session.getPcName(), severity, tabName, null);
                    }
                }
            }
        } catch (Exception e) {
            log.warn("Client {} đã ngắt kết nối: {}", session.getId(), e.getMessage());
        } finally {
            cleanup();
        }
    }

    public synchronized void sendRawCommand(String rawCommand) {
        try {
            if (out != null) {
                out.println(rawCommand);
                log.info("Đã gửi lệnh tới [{}]: {}", session.getId(), rawCommand);
            }
        } catch (Exception e) {
            log.error("Lỗi khi gửi lệnh tới Client {}: {}", session.getId(), e.getMessage());
        }
    }

    private void cleanup() {
        session.setOnline(false);
        clientManager.removeClient(session.getId());
        notificationService.notifyClientListUpdated(clientManager.getAllClients());
        try {
            if (in != null)
                in.close();
            if (out != null)
                out.close();
            if (session.getSocket() != null)
                session.getSocket().close();
        } catch (Exception ignored) {
        }
    }
}

package com.vku.lanmonitor.server.socket;

import com.vku.lanmonitor.server.model.ClientSession;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;

import java.util.ArrayList;
import java.util.List;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;

@Slf4j
@Component
public class ClientManager {

    private final Map<String, ClientSession> activeClients = new ConcurrentHashMap<>();
    private final Map<String, ClientConnectionHandler> activeHandlers = new ConcurrentHashMap<>();

    public void addClient(ClientSession session, ClientConnectionHandler handler) {
        activeClients.put(session.getId(), session);
        activeHandlers.put(session.getId(), handler);
        log.info("Đã thêm Client mới: {} - IP: {}. Tổng số Client: {}",
                session.getId(), session.getIpAddress(), activeClients.size());
    }

    public void removeClient(String clientId) {
        activeClients.remove(clientId);
        activeHandlers.remove(clientId);
        log.info("Đã ngắt kết nối Client: {}. Tổng số Client hiện tại: {}",
                clientId, activeClients.size());
    }

    public ClientSession getClient(String clientId) {
        return activeClients.get(clientId);
    }

    public ClientConnectionHandler getHandler(String clientId) {
        return activeHandlers.get(clientId);
    }

    public List<ClientSession> getAllClients() {
        return new ArrayList<>(activeClients.values());
    }

    public boolean isClientOnline(String clientId) {
        return activeClients.containsKey(clientId) && activeClients.get(clientId).isOnline();
    }

    public void broadcastCommand(String rawCommand) {
        log.info("Phát sóng lệnh tới tất cả Client: {}", rawCommand);
        for (ClientConnectionHandler handler : activeHandlers.values()) {
            handler.sendRawCommand(rawCommand);
        }
    }
}

package com.vku.lanmonitor.server.socket;

import com.vku.lanmonitor.server.model.ClientSession;
import com.vku.lanmonitor.server.websocket.RealtimeNotificationService;
import jakarta.annotation.PreDestroy;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.boot.CommandLineRunner;
import org.springframework.stereotype.Service;

import java.io.IOException;
import java.net.ServerSocket;
import java.net.Socket;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;

@Slf4j
@Service
@RequiredArgsConstructor
public class TcpServerService implements CommandLineRunner {

    private final ClientManager clientManager;
    private final RealtimeNotificationService notificationService;

    // Đọc cổng cấu hình từ application.properties (mặc định 9999)
    @Value("${lan.socket.port:9999}")
    private int port;

    private ServerSocket serverSocket;
    private boolean isRunning = false;

    // ThreadPool để quản lý luồng xử lý riêng cho từng Client kết nối vào
    private final ExecutorService threadPool = Executors.newCachedThreadPool();

    @Override
    public void run(String... args) {
        // Khởi chạy ServerSocket trong một Thread độc lập để không làm treo luồng chính
        // của Spring Boot
        new Thread(this::startServer, "TCP-Listener-Thread").start();
    }

    private void startServer() {
        try {
            serverSocket = new ServerSocket(port);
            isRunning = true;
            log.info(">>> TCP ServerSocket đã khởi động thành công trên Port: {} <<<", port);

            while (isRunning) {
                // Chờ và chấp nhận kết nối từ máy Client (hàm này sẽ block cho đến khi có máy
                // kết nối)
                Socket clientSocket = serverSocket.accept();
                String clientIp = clientSocket.getInetAddress().getHostAddress();
                log.info("Phát hiện kết nối mới từ IP: {}", clientIp);

                // Tạo một phiên kết nối mới
                ClientSession session = new ClientSession(clientIp, clientSocket);
                ClientConnectionHandler handler = new ClientConnectionHandler(
                        session, clientManager, notificationService);

                // Bàn giao kết nối cho ClientConnectionHandler xử lý trong ThreadPool
                clientManager.addClient(session, handler);
                threadPool.execute(handler);
            }
        } catch (IOException e) {
            if (isRunning) {
                log.error("Lỗi trong quá trình hoạt động của TCP ServerSocket: {}", e.getMessage());
            }
        }
    }

    @PreDestroy
    public void stopServer() {
        isRunning = false;
        try {
            if (serverSocket != null && !serverSocket.isClosed()) {
                serverSocket.close();
                log.info("Đã đóng TCP ServerSocket an toàn.");
            }
            threadPool.shutdown();
        } catch (IOException e) {
            log.error("Lỗi khi đóng TCP ServerSocket: {}", e.getMessage());
        }
    }
}

package com.vku.lanmonitor.server.websocket;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.messaging.simp.SimpMessagingTemplate;
import org.springframework.stereotype.Service;

@Slf4j
@Service
@RequiredArgsConstructor
public class RealtimeNotificationService {

    private final SimpMessagingTemplate messagingTemplate;

    /**
     * Đẩy thông báo cập nhật danh sách Client online lên Web
     */
    public void notifyClientListUpdated(Object clients) {
        messagingTemplate.convertAndSend("/topic/clients", clients);
        log.info("Đã đẩy danh sách Client mới lên Web qua WebSocket.");
    }

    /**
     * Đẩy ảnh chụp màn hình nhận được từ máy con lên Web
     */
    public void sendScreenshotToWeb(String clientId, String base64Image) {
        messagingTemplate.convertAndSend("/topic/screenshot/" + clientId, base64Image);
        log.info("Đã đẩy ảnh chụp màn hình của client [{}] lên Web.", clientId);
    }

    public void sendSuspiciousAlertToWeb(String clientId, String pcName, String severity, String tabName,
            String imageUrl) {
        java.util.Map<String, String> alertData = new java.util.HashMap<>();
        alertData.put("clientId", clientId);
        alertData.put("pcName", pcName != null ? pcName : "Máy Trạm");
        alertData.put("severity", severity);
        alertData.put("tabName", tabName);
        alertData.put("imageUrl", imageUrl); // Link ảnh để Web load

function connectWebSocket() {
  const socket = new SockJS("/ws");
  stompClient = Stomp.over(socket);
  stompClient.debug = null; // Tắt bớt log debug STOMP trong console

  stompClient.connect(
    {},
    function () {
      const statusEl = document.getElementById("socket-status");
      statusEl.innerText = "WebSocket Đã Kết Nối";
      statusEl.className =
        "px-3 py-1 text-xs rounded-full bg-emerald-500/10 text-emerald-400 border border-emerald-500/20";

      // Lắng nghe cập nhật danh sách máy con
      stompClient.subscribe("/topic/clients", function (message) {
        const clients = JSON.parse(message.body);
        renderClientList(clients);
      });

      // Lắng nghe cảnh báo gian lận (ĐÃ ĐƯỢC ĐƯA VÀO ĐÚNG VỊ TRÍ)
      stompClient.subscribe("/topic/alerts", function (message) {

let liveStreamInterval = null;

function openLiveStream(clientId, pcName) {
  activeLiveStreamClient = clientId;
  document.getElementById("livestream-title").innerText = pcName;

  const modal = document.getElementById("livestream-modal");
  modal.classList.remove("hidden");
  modal.classList.add("flex");

  const topic = `/topic/screenshot/${clientId}`;
  if (!currentSubscriptions[topic] && stompClient) {
    currentSubscriptions[topic] = stompClient.subscribe(topic, function (msg) {
      if (activeLiveStreamClient === clientId) {
        document.getElementById("livestream-img").src =
          "data:image/jpeg;base64," + msg.body;
      }
    });
  }

  if (liveStreamInterval) clearInterval(liveStreamInterval);
  liveStreamInterval = setInterval(() => {
    fetch(`/api/clients/${encodeURIComponent(clientId)}/capture`, {
      method: "POST",
    }).catch((e) => console.log("Lỗi Livestream: ", e));
  }, 250);
}

async function saveWhitelist() {
  const inputStr = document.getElementById("whitelist-input").value;
  // Cắt chuỗi bằng dấu phẩy, loại bỏ khoảng trắng thừa
  const whitelistArray = inputStr
    .split(",")
    .map((item) => item.trim())
    .filter((item) => item.length > 0);

  if (whitelistArray.length === 0) {
    alert("Vui lòng nhập ít nhất một từ khóa hợp lệ!");
    return;
  }

  try {
    const res = await fetch("/api/settings/whitelist", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ whitelist: whitelistArray }),
    });

    if (res.ok) {
      alert(
        "✅ Đã lưu cấu hình và cập nhật luật mới xuống tất cả máy tính đang thi!",
      );
      closeWhitelistModal();
    }
  } catch (e) {
    alert("Lỗi khi lưu cấu hình: " + e.message);
  }
}

    function escapeHtml(value) {
      return String(value ?? "")
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;")
        .replace(/'/g, "&#39;");
    }
